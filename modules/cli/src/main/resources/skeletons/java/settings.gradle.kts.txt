pluginManagement {
    repositories {
        mavenCentral()
        gradlePluginPortal()
    }
}

plugins {
    id("org.gradle.toolchains.foojay-resolver-convention") version "0.8.0"
}

rootProject.name = "{{appName}}"

// Build against a local Kona checkout instead of published artifacts:
//   ./gradlew run -Pkona.home=/path/to/kona   (or export KONA_HOME=/path/to/kona)
val konaHome = providers.gradleProperty("kona.home")
    .orElse(providers.environmentVariable("KONA_HOME"))
    .orNull
if (konaHome != null) {
    includeBuild(konaHome)
}
