#!/usr/bin/env bash
# App-local grind — Termwind-styled CLI. Gradle stays quiet; only grind output shows.
set -euo pipefail
DIR=$(CDPATH= cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)
cd "$DIR"

# Quiet + plain: no lifecycle spam / stuck ~83% progress while `run` (serve) lives.
GRADLE=(./gradlew --quiet --console=plain -Dorg.gradle.console=plain)

# Resolve the app classpath, then exec the JVM so stdin stays a TTY.
# Gradle JavaExec workers typically see a pipe, which disables boxed prompts
# (Yes/No arrows, choice lists). Fall back to the grind JavaExec task if
# konaClasspath cannot run.
CLASSPATH="$("${GRADLE[@]}" konaClasspath)" || CLASSPATH=""
if [[ -n "$CLASSPATH" ]]; then
  JAVA="${JAVA_HOME:+$JAVA_HOME/bin/java}"
  JAVA="${JAVA:-java}"
  exec "$JAVA" -cp "$CLASSPATH" org.konaframework.grind.GrindKt "$@"
fi

if [[ $# -eq 0 ]]; then
  exec "${GRADLE[@]}" grind
fi
exec "${GRADLE[@]}" grind -PgrindArgs="$*"
