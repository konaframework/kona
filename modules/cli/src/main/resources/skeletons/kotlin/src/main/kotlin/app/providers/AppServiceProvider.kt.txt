package app.providers

import app.models.User
import org.konaframework.auth.AuthManager
import org.konaframework.auth.CadenceUserProvider
import org.konaframework.auth.SessionGuard
import org.konaframework.auth.hashing.HashManager
import org.konaframework.core.Application
import org.konaframework.core.ServiceProvider

class AppServiceProvider : ServiceProvider {
    override fun register(app: Application) {
        // Bind application services here.
        // Migrations from grind make:migration are auto-loaded via META-INF/kona/migrations.
    }

    override fun boot(app: Application) {
        val auth = app.resolve(AuthManager::class.java)
        val hasher = app.resolve(HashManager::class.java).driver()
        val users = CadenceUserProvider(User::class.java, hasher)
        auth.provider("users", users)
        auth.extend("web", SessionGuard("web", users))
    }
}
