package bootstrap

import app.http.middleware.HandleInertia
import app.providers.AppServiceProvider
import config.App as AppConfig
import config.Database
import config.Filesystems
import config.Http
import config.Paths
import config.Session
import io.konaframework.auth.AuthServiceProvider
import io.konaframework.core.Application
import io.konaframework.core.Env
import io.konaframework.core.Kona
import io.konaframework.crema.CremaServiceProvider
import io.konaframework.cadence.CadenceServiceProvider
import io.konaframework.filesystem.FilesystemServiceProvider
import io.konaframework.http.HttpKernel
import io.konaframework.http.HttpServiceProvider
import io.konaframework.http.KonaHttpServer
import io.konaframework.http.Router
import io.konaframework.inertia.InertiaServiceProvider
import io.konaframework.mail.MailServiceProvider
import io.konaframework.queue.QueueServiceProvider
import io.konaframework.redis.RedisServiceProvider
import io.konaframework.schedule.ScheduleServiceProvider
import io.konaframework.session.SessionServiceProvider
import routes.ApiRoutes
import routes.WebRoutes

object App {
    @JvmStatic
    fun create(args: Array<String>): Application {
        val app = Kona.create(args)
            .configs(AppConfig, Database, Session, Filesystems, Http, Paths)
            .configure { cfg ->
                cfg.set("inertia.root_view", "app")
                cfg.set("inertia.version", "1")
            }
            .providers(
                HttpServiceProvider(),
                CadenceServiceProvider(),
                FilesystemServiceProvider(),
                RedisServiceProvider(),
                SessionServiceProvider(),
                AuthServiceProvider(),
                QueueServiceProvider(),
                ScheduleServiceProvider(),
                MailServiceProvider(),
                CremaServiceProvider(),
                InertiaServiceProvider(),
                AppServiceProvider(),
            )
            .afterBoot { application ->
                Middleware.configure(application)
                application.resolve(HttpKernel::class.java).pushMiddleware(HandleInertia())
            }
            .boot()
        WebRoutes().register(app.resolve(Router::class.java))
        ApiRoutes().register(app.resolve(Router::class.java))
        return app
    }

    @JvmStatic
    fun main(args: Array<String>) {
        val app = create(args)
        val host = Env.getOrDefault("HOST", "127.0.0.1")
        val port = Env.getOrDefault("PORT", "4000").toInt()
        KonaHttpServer(app.resolve(HttpKernel::class.java), host, port).start()
    }
}
