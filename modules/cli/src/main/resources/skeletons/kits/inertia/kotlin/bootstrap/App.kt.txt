package bootstrap

import app.Http.Middleware.HandleInertia
import app.Providers.AppServiceProvider
import dev.kona.core.Application
import dev.kona.core.Env
import dev.kona.core.Kona
import dev.kona.crema.CremaServiceProvider
import dev.kona.http.HttpKernel
import dev.kona.http.HttpServiceProvider
import dev.kona.http.KonaHttpServer
import dev.kona.http.Router
import dev.kona.inertia.InertiaServiceProvider
import routes.WebRoutes

/**
 * Application bootstrap. `grind serve --in-process` calls [create]; `./gradlew run` calls [main].
 */
object App {
    @JvmStatic
    fun create(args: Array<String>): Application {
        val app = Kona.create(args)
            .configure { cfg ->
                cfg.set("inertia.root_view", "app")
                cfg.set("inertia.version", "1")
            }
            .providers(
                HttpServiceProvider(),
                CremaServiceProvider(),
                InertiaServiceProvider(),
                AppServiceProvider(),
            )
            .boot()
        app.resolve(HttpKernel::class.java).pushMiddleware(HandleInertia())
        WebRoutes().register(app.resolve(Router::class.java))
        return app
    }

    @JvmStatic
    fun main(args: Array<String>) {
        val app = create(args)
        val host = Env.getOrDefault("HOST", "127.0.0.1")
        val port = Env.getOrDefault("PORT", "8080").toInt()
        KonaHttpServer(app.resolve(HttpKernel::class.java), host, port).start()
    }
}
