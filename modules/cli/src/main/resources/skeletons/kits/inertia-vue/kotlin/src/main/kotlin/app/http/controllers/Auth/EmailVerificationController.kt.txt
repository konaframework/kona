package app.http.controllers.Auth
import app.models.User
import org.konaframework.auth.Auth
import org.konaframework.auth.MustVerifyEmail
import org.konaframework.http.Request
import org.konaframework.http.Response
import org.konaframework.http.Url
import org.konaframework.inertia.Inertia
class EmailVerificationController {
    fun notice(request: Request): Response {
        Auth.setRequest(request)
        val user = Auth.user() as? MustVerifyEmail
        if (user != null && user.hasVerifiedEmail()) return Response.redirect("/dashboard", 302)
        return Inertia.render("Auth/VerifyEmail", mapOf("status" to request.session()?.get("status")))
    }
    fun verify(request: Request): Response {
        Auth.setRequest(request)
        val user = Auth.user() as? User ?: return Response.redirect("/login", 302)
        if (!Url.generator().hasValidSignature(request)) return Response.redirect("/email/verify", 302)
            .withErrors(mapOf("email" to listOf("Invalid or expired verification link.")))
        if (!user.hasVerifiedEmail()) user.markEmailAsVerified()
        return Response.redirect("/dashboard", 302).with("status", "Your email has been verified.")
    }
    fun send(request: Request): Response {
        Auth.setRequest(request)
        val user = Auth.user() as? User ?: return Response.redirect("/login", 302)
        if (user.hasVerifiedEmail()) return Response.redirect("/dashboard", 302)
        val link = Url.generator().signedRoute("verification.verify", mapOf(
            "id" to user.getAuthIdentifier().toString(),
            "hash" to user.getEmailForVerification().hashCode().toString(),
        ), expirationSeconds = 3600)
        return Response.redirect("/email/verify", 302).with("status", "Verification link ready (local): $link")
    }
}
