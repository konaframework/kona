package app.http.controllers.Auth
import org.konaframework.auth.Auth
import org.konaframework.auth.HasEmail
import org.konaframework.auth.middleware.Authenticate
import org.konaframework.auth.middleware.RequirePassword
import org.konaframework.http.Request
import org.konaframework.http.Response
import org.konaframework.inertia.Inertia
class ConfirmPasswordController {
    fun show(request: Request): Response = Inertia.render("Auth/ConfirmPassword", emptyMap())
    fun store(request: Request): Response {
        Auth.setRequest(request)
        val password = request.input("password")?.toString().orEmpty()
        val user = Auth.user() ?: return Response.redirect("/login", 302)
        val email = (user as? HasEmail)?.getEmail()
        if (!Auth.validate(mapOf("email" to email, "password" to password))) {
            return Response.redirect("/user/confirm-password", 302)
                .withErrors(mapOf("password" to listOf("The provided password is incorrect.")))
        }
        RequirePassword.confirm(request)
        return Response.redirect(Authenticate.pullIntended(request, "/dashboard"), 302)
    }
}
