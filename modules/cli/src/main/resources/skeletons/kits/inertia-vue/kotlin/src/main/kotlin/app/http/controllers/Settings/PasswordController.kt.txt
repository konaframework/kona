package app.http.controllers.Settings
import app.models.User
import org.konaframework.auth.Auth
import org.konaframework.http.Request
import org.konaframework.http.Response
class PasswordController {
    fun update(request: Request): Response {
        Auth.setRequest(request)
        val user = Auth.user() as User
        val current = request.input("current_password")?.toString().orEmpty()
        val password = request.input("password")?.toString().orEmpty()
        val confirmation = request.input("password_confirmation")?.toString().orEmpty()
        val errors = linkedMapOf<String, List<String>>()
        if (!Auth.validate(mapOf("email" to user.getEmail(), "password" to current)))
            errors["current_password"] = listOf("The provided password is incorrect.")
        if (password.length < 8) errors["password"] = listOf("The password must be at least 8 characters.")
        if (password != confirmation) errors["password"] = listOf("The password confirmation does not match.")
        if (errors.isNotEmpty()) return Response.redirect("/settings/security", 302).withErrors(errors)
        user.setAttribute("password", password); user.save()
        return Response.redirect("/settings/security", 302).with("status", "Password updated.")
    }
}
