package app.http.controllers.settings;
import app.models.User; import org.konaframework.auth.Auth; import org.konaframework.http.Request; import org.konaframework.http.Response;
import java.util.LinkedHashMap; import java.util.List; import java.util.Map;
public class PasswordController {
  public Response update(Request request) {
    Auth.setRequest(request); User user = (User) Auth.user();
    String current = request.input("current_password")==null?"":request.input("current_password");
    String password = request.input("password")==null?"":request.input("password");
    String confirmation = request.input("password_confirmation")==null?"":request.input("password_confirmation");
    Map<String, List<String>> errors = new LinkedHashMap<>();
    if (!Auth.validate(Map.of("email", user.getEmail(), "password", current))) errors.put("current_password", List.of("The provided password is incorrect."));
    if (password.length()<8) errors.put("password", List.of("The password must be at least 8 characters."));
    if (!password.equals(confirmation)) errors.put("password", List.of("The password confirmation does not match."));
    if (!errors.isEmpty()) return Response.redirect("/settings/security",302).withErrors(errors);
    user.setAttribute("password", password); user.save();
    return Response.redirect("/settings/security",302).with("status","Password updated.");
  }
}
