package app.http.controllers.Auth
import app.models.User
import org.konaframework.auth.passwords.PasswordBroker
import org.konaframework.auth.passwords.PasswordResetResult
import org.konaframework.http.Request
import org.konaframework.http.Response
import org.konaframework.inertia.Inertia
class PasswordResetController(private val passwords: PasswordBroker) {
    fun create(request: Request): Response = Inertia.render("Auth/ForgotPassword", mapOf("status" to request.session()?.get("status")))
    fun store(request: Request): Response {
        val email = request.input("email")?.toString().orEmpty().trim()
        if (email.isBlank()) return Response.redirect("/forgot-password", 302).withErrors(mapOf("email" to listOf("The email field is required.")))
        passwords.createToken(mapOf("email" to email))
        return Response.redirect("/forgot-password", 302).with("status", "We have emailed your password reset link.")
    }
    fun edit(request: Request): Response = Inertia.render("Auth/ResetPassword", mapOf(
        "token" to (request.query["token"] ?: request.input("token") ?: ""),
        "email" to (request.query["email"] ?: request.input("email") ?: ""),
    ))
    fun update(request: Request): Response {
        val email = request.input("email")?.toString().orEmpty()
        val token = request.input("token")?.toString().orEmpty()
        val password = request.input("password")?.toString().orEmpty()
        val confirmation = request.input("password_confirmation")?.toString().orEmpty()
        if (password.length < 8 || password != confirmation) return Response.redirect("/reset-password", 302)
            .withErrors(mapOf("password" to listOf("The password must be at least 8 characters and match confirmation.")))
            .withInput(mapOf("email" to email, "token" to token))
        val result = passwords.reset(mapOf("email" to email, "token" to token, "password" to password)) { user, newPassword ->
            (user as User).setAttribute("password", newPassword); (user as User).save()
        }
        return if (result == PasswordResetResult.RESET) Response.redirect("/login", 302).with("status", "Your password has been reset.")
        else Response.redirect("/reset-password", 302).withErrors(mapOf("email" to listOf("This password reset token is invalid.")))
    }
}
