package app.http.controllers.Settings
import app.models.User
import app.support.Features
import org.konaframework.auth.Auth
import org.konaframework.auth.twofactor.ConfirmTwoFactorAuthentication
import org.konaframework.auth.twofactor.DisableTwoFactorAuthentication
import org.konaframework.auth.twofactor.EnableTwoFactorAuthentication
import org.konaframework.auth.twofactor.GenerateNewRecoveryCodes
import org.konaframework.auth.twofactor.TwoFactor
import org.konaframework.core.Env
import org.konaframework.http.Request
import org.konaframework.http.Response
import org.konaframework.inertia.Inertia
import org.konaframework.security.Encrypter
class TwoFactorController(
    private val enable: EnableTwoFactorAuthentication,
    private val confirm: ConfirmTwoFactorAuthentication,
    private val disable: DisableTwoFactorAuthentication,
    private val regenerate: GenerateNewRecoveryCodes,
    private val encrypter: Encrypter,
) {
    fun edit(request: Request): Response {
        Auth.setRequest(request)
        val user = Auth.user() as User
        val props = linkedMapOf<String, Any?>(
            "enabled" to (Features.twoFactorAuthentication() && user.hasEnabledTwoFactorAuthentication()),
            "requiresConfirmation" to TwoFactor.requiresConfirmation(),
            "status" to request.session()?.get("status"),
        )
        if (!user.getTwoFactorSecret().isNullOrBlank() && Features.twoFactorAuthentication()) {
            props["qrCodeUrl"] = user.twoFactorQrCodeUrl(TwoFactor.provider(), encrypter, Env.getOrDefault("APP_NAME", "Kona"), user.getEmail().orEmpty())
            props["recoveryCodes"] = user.recoveryCodes(encrypter)
            props["confirmed"] = user.getTwoFactorConfirmedAt() != null
        }
        return Inertia.render("Settings/Security", props)
    }
    fun enable(request: Request): Response {
        if (!Features.twoFactorAuthentication()) return Response.redirect("/settings/security", 302)
        Auth.setRequest(request); enable.enable(Auth.user() as User, force = true)
        return Response.redirect("/settings/security", 302).with("status", "Two-factor authentication enabled. Confirm with an authenticator code.")
    }
    fun confirm(request: Request): Response {
        Auth.setRequest(request)
        if (!confirm.confirm(Auth.user() as User, request.input("code")?.toString().orEmpty())) {
            return Response.redirect("/settings/security", 302).withErrors(mapOf("code" to listOf("The provided two-factor authentication code was invalid.")))
        }
        return Response.redirect("/settings/security", 302).with("status", "Two-factor authentication confirmed.")
    }
    fun disable(request: Request): Response {
        Auth.setRequest(request); disable.disable(Auth.user() as User)
        return Response.redirect("/settings/security", 302).with("status", "Two-factor authentication disabled.")
    }
    fun recoveryCodes(request: Request): Response {
        Auth.setRequest(request); regenerate.generate(Auth.user() as User)
        return Response.redirect("/settings/security", 302).with("status", "Recovery codes regenerated.")
    }
}
