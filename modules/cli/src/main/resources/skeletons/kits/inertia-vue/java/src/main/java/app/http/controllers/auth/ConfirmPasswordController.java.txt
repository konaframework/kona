package app.http.controllers.auth;
import org.konaframework.auth.Auth; import org.konaframework.auth.HasEmail; import org.konaframework.auth.middleware.Authenticate; import org.konaframework.auth.middleware.RequirePassword;
import org.konaframework.http.Request; import org.konaframework.http.Response; import org.konaframework.inertia.Inertia;
import java.util.List; import java.util.Map;
public class ConfirmPasswordController {
  public Response show(Request request) { return Inertia.render("Auth/ConfirmPassword", Map.of()); }
  public Response store(Request request) {
    Auth.setRequest(request);
    String password = request.input("password")==null?"":request.input("password");
    var user = Auth.user();
    if (user == null) return Response.redirect("/login",302);
    String email = user instanceof HasEmail e ? e.getEmail() : null;
    if (!Auth.validate(Map.of("email", email, "password", password))) {
      return Response.redirect("/user/confirm-password",302).withErrors(Map.of("password", List.of("The provided password is incorrect.")));
    }
    RequirePassword.confirm(request);
    return Response.redirect(Authenticate.pullIntended(request, "/dashboard"),302);
  }
}
