package app.http.controllers.auth;
import org.konaframework.auth.passwords.PasswordBroker;
import org.konaframework.auth.passwords.PasswordResetResult;
import app.models.User;
import org.konaframework.http.Request; import org.konaframework.http.Response; import org.konaframework.inertia.Inertia;
import java.util.List; import java.util.Map;
public class PasswordResetController {
  private final PasswordBroker passwords;
  public PasswordResetController(PasswordBroker passwords) { this.passwords = passwords; }
  public Response create(Request request) { return Inertia.render("Auth/ForgotPassword", Map.of("status", request.session()==null?null:request.session().get("status"))); }
  public Response store(Request request) {
    String email = request.input("email")==null?"":request.input("email").trim();
    if (email.isBlank()) return Response.redirect("/forgot-password",302).withErrors(Map.of("email", List.of("The email field is required.")));
    passwords.createToken(Map.of("email", email));
    return Response.redirect("/forgot-password",302).with("status","We have emailed your password reset link.");
  }
  public Response edit(Request request) {
    String token = request.getQuery().getOrDefault("token", request.input("token") == null ? "" : request.input("token"));
    String email = request.getQuery().getOrDefault("email", request.input("email") == null ? "" : request.input("email"));
    return Inertia.render("Auth/ResetPassword", Map.of("token", token, "email", email));
  }
  public Response update(Request request) {
    String email=request.input("email")==null?"":request.input("email");
    String token=request.input("token")==null?"":request.input("token");
    String password=request.input("password")==null?"":request.input("password");
    String confirmation=request.input("password_confirmation")==null?"":request.input("password_confirmation");
    if (password.length()<8 || !password.equals(confirmation)) {
      return Response.redirect("/reset-password",302).withErrors(Map.of("password", List.of("The password must be at least 8 characters and match confirmation."))).withInput(Map.of("email", email, "token", token));
    }
    PasswordResetResult result = passwords.reset(Map.of("email", email, "token", token, "password", password), (user, newPassword) -> {
      User model = (User) user; model.setAttribute("password", newPassword); model.save();
    });
    if (result == PasswordResetResult.RESET) return Response.redirect("/login",302).with("status","Your password has been reset.");
    return Response.redirect("/reset-password",302).withErrors(Map.of("email", List.of("This password reset token is invalid.")));
  }
}
