package bootstrap

import app.providers.AppServiceProvider
import config.App as AppConfig
import config.Auth
import config.Broadcasting
import config.Cache
import config.Database
import config.Filesystems
import config.Hash
import config.Http
import config.Logging
import config.Mail
import config.Paths
import config.Queue
import config.Redis
import config.Services
import config.Session
import org.konaframework.auth.AuthServiceProvider
import org.konaframework.broadcasting.BroadcastServiceProvider
import org.konaframework.cache.CacheServiceProvider
import org.konaframework.core.Application
import org.konaframework.core.Env
import org.konaframework.core.Kona
import org.konaframework.crema.CremaServiceProvider
import org.konaframework.cadence.CadenceServiceProvider
import org.konaframework.filesystem.FilesystemServiceProvider
import org.konaframework.http.HttpKernel
import org.konaframework.http.HttpServiceProvider
import org.konaframework.http.KonaHttpServer
import org.konaframework.http.Router
import org.konaframework.mail.MailServiceProvider
import org.konaframework.queue.QueueServiceProvider
import org.konaframework.redis.RedisServiceProvider
import org.konaframework.schedule.ScheduleServiceProvider
import org.konaframework.session.SessionServiceProvider
import routes.ApiRoutes
import routes.WebRoutes

object App {
    @JvmStatic
    fun create(args: Array<String>): Application {
        val app = Kona.create(args)
            .configs(
                AppConfig, Database, Redis, Session, Cache, Filesystems, Queue, Mail,
                Logging, Auth, Hash, Broadcasting, Services, Http, Paths,
            )
            .providers(
                HttpServiceProvider(),
                CadenceServiceProvider(),
                FilesystemServiceProvider(),
                RedisServiceProvider(),
                CacheServiceProvider(),
                SessionServiceProvider(),
                AuthServiceProvider(),
                QueueServiceProvider(),
                ScheduleServiceProvider(),
                MailServiceProvider(),
                BroadcastServiceProvider(),
                CremaServiceProvider(),
                AppServiceProvider(),
            )
            .afterBoot { Middleware.configure(it) }
            .boot()
        val router = app.resolve(Router::class.java)
        WebRoutes().register(router)
        ApiRoutes().register(router)
        return app
    }

    @JvmStatic
    fun main(args: Array<String>) {
        val app = create(args)
        val host = Env.getOrDefault("HOST", "127.0.0.1")
        val port = Env.getOrDefault("PORT", "4000").toInt()
        KonaHttpServer(app.resolve(HttpKernel::class.java), host, port).start()
    }
}
