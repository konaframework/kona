package app.Http.Controllers

import dev.kona.brew.Brew
import dev.kona.brew.BrewAssets
import dev.kona.http.Request
import dev.kona.http.Response

class WelcomeController {
    fun index(request: Request): Response {
        val body =
            """
            <!DOCTYPE html>
            <html lang="en">
              <head>
                <meta charset="utf-8" />
                <title>Brew</title>
                ${BrewAssets.scriptTag()}
              </head>
              <body>
                <h1>Welcome to Kona Brew</h1>
                ${Brew.mount("counter")}
              </body>
            </html>
            """.trimIndent()
        return Response.html(body)
    }
}
