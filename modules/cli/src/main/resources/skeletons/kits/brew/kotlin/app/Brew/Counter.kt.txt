package app.Brew

import dev.kona.brew.Brew
import dev.kona.brew.Component
import dev.kona.core.Kona
import dev.kona.crema.CremaEngine

class Counter : Component() {
    @JvmField
    var count: Int = 0

    fun increment() {
        count++
    }

    fun decrement() {
        count--
    }

    override fun render(): String {
        val crema = Kona.app().resolve(CremaEngine::class.java)
        return Brew.view(crema, "brew-counter", mapOf("count" to count))
    }
}
