package bootstrap

import app.brew.Counter
import app.providers.AppServiceProvider
import config.App as AppConfig
import config.Database
import config.Filesystems
import config.Http
import config.Paths
import config.Session
import dev.kona.auth.AuthServiceProvider
import dev.kona.brew.Brew
import dev.kona.brew.BrewServiceProvider
import dev.kona.core.Application
import dev.kona.core.Env
import dev.kona.core.Kona
import dev.kona.crema.CremaServiceProvider
import dev.kona.extract.ExtractServiceProvider
import dev.kona.filesystem.FilesystemServiceProvider
import dev.kona.http.HttpKernel
import dev.kona.http.HttpServiceProvider
import dev.kona.http.KonaHttpServer
import dev.kona.http.Router
import dev.kona.mail.MailServiceProvider
import dev.kona.queue.QueueServiceProvider
import dev.kona.redis.RedisServiceProvider
import dev.kona.schedule.ScheduleServiceProvider
import dev.kona.session.SessionServiceProvider
import routes.ApiRoutes
import routes.WebRoutes

object App {
    @JvmStatic
    fun create(args: Array<String>): Application {
        val app = Kona.create(args)
            .configs(AppConfig, Database, Session, Filesystems, Http, Paths)
            .providers(
                HttpServiceProvider(),
                ExtractServiceProvider(),
                FilesystemServiceProvider(),
                RedisServiceProvider(),
                SessionServiceProvider(),
                AuthServiceProvider(),
                QueueServiceProvider(),
                ScheduleServiceProvider(),
                MailServiceProvider(),
                CremaServiceProvider(),
                BrewServiceProvider(),
                AppServiceProvider(),
            )
            .afterBoot { Middleware.configure(it) }
            .boot()
        // Pin the sample counter; other app.brew.* classes resolve via brew.class_namespace.
        Brew.component(Counter::class.java)
        WebRoutes().register(app.resolve(Router::class.java))
        ApiRoutes().register(app.resolve(Router::class.java))
        return app
    }

    @JvmStatic
    fun main(args: Array<String>) {
        val app = create(args)
        val host = Env.getOrDefault("HOST", "127.0.0.1")
        val port = Env.getOrDefault("PORT", "4000").toInt()
        KonaHttpServer(app.resolve(HttpKernel::class.java), host, port).start()
    }
}
